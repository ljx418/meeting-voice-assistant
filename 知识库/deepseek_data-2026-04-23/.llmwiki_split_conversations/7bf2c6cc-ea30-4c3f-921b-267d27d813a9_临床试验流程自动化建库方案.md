import pandas as pd
from docx import Document
import PyPDF2

def process_uploaded_files(context):
    """
    处理上传的文件
    context: 包含上传文件信息的上下文
    """
    
    # 获取文件路径（Dify会将上传的文件保存到临时路径）
    trial_file_path = context.get("trial_protocol_file")
    als_file_path = context.get("als_template_file")
    
    results = {
        "trial_data": {},
        "als_data": {},
        "extracted_info": {}
    }
    
    # 根据文件类型解析
    if trial_file_path.endswith('.pdf'):
        # 解析PDF格式的试验流程表
        trial_text = extract_pdf_content(trial_file_path)
        results["trial_data"]["text"] = trial_text
        results["trial_data"]["visits"] = extract_visits_from_text(trial_text)
        
    elif trial_file_path.endswith('.docx'):
        # 解析Word文档
        trial_text = extract_docx_content(trial_file_path)
        results["trial_data"]["text"] = trial_text
        
    elif trial_file_path.endswith(('.xlsx', '.xls')):
        # 解析Excel文件
        trial_df = pd.read_excel(trial_file_path)
        results["trial_data"]["dataframe"] = trial_df.to_dict()
        results["trial_data"]["visits"] = extract_visits_from_excel(trial_df)
    
    # 解析ALS模板文件（通常是Excel）
    if als_file_path.endswith(('.xlsx', '.xls')):
        als_df = pd.read_excel(als_file_path)
        results["als_data"] = {
            "forms": extract_forms_from_als(als_df),
            "domains": extract_domains_from_als(als_df)
        }
    
    # 提取访视信息
    if "visits" in results["trial_data"]:
        results["extracted_info"]["visits"] = results["trial_data"]["visits"]
    
    # 匹配表单和域名
    if results["trial_data"] and results["als_data"]:
        matches = match_forms_to_als(
            results["trial_data"]["visits"], 
            results["als_data"]["forms"]
        )
        results["extracted_info"]["matches"] = matches
    
    return results

def extract_visits_from_excel(df):
    """从Excel中提取访视信息"""
    visits = []
    # 假设访视信息在名为"访视计划"的工作表
    if '访视计划' in df.sheet_names:
        visits_df = pd.read_excel(df, sheet_name='访视计划')
        for _, row in visits_df.iterrows():
            visit = {
                "visit_name": row.get("访视名称"),
                "time_window": row.get("时间窗"),
                "forms": row.get("评估表单").split(",") if row.get("评估表单") else []
            }
            visits.append(visit)
    return visits

def match_forms_to_als(trial_forms, als_forms):
    """匹配试验表单到ALS标准表单"""
    matches = []
    for form in trial_forms:
        # 模糊匹配逻辑
        best_match = find_best_match(form, als_forms)
        matches.append({
            "trial_form": form,
            "als_form": best_match["name"],
            "domain": best_match["domain"],
            "confidence": best_match["confidence"]
        })
    return matches
