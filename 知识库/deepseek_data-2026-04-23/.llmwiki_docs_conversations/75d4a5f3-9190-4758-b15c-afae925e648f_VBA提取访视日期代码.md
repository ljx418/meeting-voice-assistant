Function CleanString(ByVal text As String) As String
    If IsNull(text) Or text = "" Then
        CleanString = ""
        Exit Function
    End If
    
    Dim result As String
    result = text
    
    ' 替换不间断空格（ASCII 160 或 Unicode \xa0）
    result = Replace(result, Chr(160), " ")     ' Chr(160) 就是不间断空格
    result = Replace(result, ChrW(160), " ")    ' Unicode 版本
    
    ' 替换其他常见特殊空格字符
    result = Replace(result, Chr(9), " ")       ' 制表符
    result = Replace(result, Chr(10), " ")      ' 换行符
    result = Replace(result, Chr(13), " ")      ' 回车符
    
    ' 替换多个连续空格为单个空格
    Do While InStr(result, "  ") > 0
        result = Replace(result, "  ", " ")
    Loop
    
    ' 去除首尾空格
    result = Trim(result)
    
    CleanString = result
End Function

Function CleanString(ByVal text As String) As String
    If text = "" Then
        CleanString = ""
        Exit Function
    End If
    
    Dim result As String
    result = text

    result = Replace(result, ChrW(160), " ")

    result = Replace(result, Chr(10), " ")
    result = Replace(result, Chr(13), " ")

    Done
    
    CleanString = Trim(result)

Function AnalyzeAndValidateSheet(ws As Worksheet, dict As Object, wsOutput As Worksheet, _
                                  ByRef outputRow As Long, ByRef errorCount As Long) As Variant
    Dim result(6) As Variant
    Dim lastRow As Long
    Dim lastCol As Long
    Dim wsData As Variant
    Dim subjectCol As Long
    Dim folderCol As Long
    Dim dateColIndices As Collection
    Dim dateColNames As Collection
    Dim j As Long
    Dim hasVisitColumns As Boolean
    Dim colOffset As Long
    Dim dateColList As String

    ' 获取数据范围
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    result(1) = lastRow - 1
    result(2) = lastCol

    If lastRow < 2 Then
        result(0) = "未检查：无数据行"
        result(3) = "-"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 批量读取数据
    On Error Resume Next
    wsData = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, lastCol)).Value
    If Err.Number <> 0 Then
        result(0) = "未检查：读取错误"
        result(3) = Err.Description
        AnalyzeAndValidateSheet = result
        Exit Function
    End If
    On Error GoTo 0

    ' 查找Subject列
    subjectCol = FindColumnBilingual(wsData, "subject", 1, lastCol)
    If subjectCol = 0 Then
        result(0) = "未检查：未找到Subject列"
        result(3) = "未找到：Subject/受试者代码/受试者姓名或标识等列名"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 查找Folder列
    folderCol = FindColumnBilingual(wsData, "folder", 1, lastCol)
    If folderCol = 0 Then
        result(0) = "未检查：未找到Folder列"
        result(3) = "未找到：Folder instance name/访视名称等列名"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 检查是否已添加访视日期列
    hasVisitColumns = (InStr(CStr(wsData(1, 1)), "Screening") > 0 And InStr(CStr(wsData(1, 1)), "Date") > 0)
    If hasVisitColumns Then
        colOffset = 3
    Else
        colOffset = 0
    End If

    ' 查找日期列
    Set dateColIndices = New Collection
    Set dateColNames = New Collection

    For j = 1 To lastCol
        If hasVisitColumns And j <= 3 Then GoTo NextColumn

        Dim colName As String
        colName = CleanString(Trim(CStr(wsData(1, j))))

        Dim hasDateKeyword As Boolean
        hasDateKeyword = False

        For Each Key In dateColKeywords.Keys
            If InStr(1, colName, Key, vbTextCompare) > 0 Then
                If InStr(1, LCase(colName), "update") > 0 Or _
                   InStr(1, LCase(colName), "creation") > 0 Or _
                   InStr(1, LCase(colName), "更新") > 0 Or _
                   InStr(1, LCase(colName), "创建") > 0 Then
                    hasDateKeyword = False
                Else
                    hasDateKeyword = True
                    Exit For
                End If
            End If
        Next

        If hasDateKeyword Then
            dateColIndices.Add j
            dateColNames.Add colName
        End If

NextColumn:
    Next j

    If dateColIndices.Count = 0 Then
        result(0) = "未检查：未找到日期列"
        result(3) = "未找到包含Date/日期关键词的列（已排除update/creation等列）"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 构建日期列列表
    dateColList = ""
    For j = 1 To WorksheetFunction.Min(dateColNames.Count, 3)
        If j > 1 Then dateColList = dateColList & ", "
        dateColList = dateColList & dateColNames(j)
    Next j
    If dateColNames.Count > 3 Then
        dateColList = dateColList & " ... (共" & dateColNames.Count & "列)"
    End If

    ' 执行验证
    Dim sheetErrorCount As Long
    sheetErrorCount = ValidateSheetData(ws, wsData, dict, wsOutput, outputRow, _
                                        lastRow, subjectCol, folderCol, dateColIndices, dateColNames, colOffset)

    result(4) = dateColList
    result(5) = "已完成检查"
    result(6) = sheetErrorCount

    AnalyzeAndValidateSheet = result
End Function

' 验证Sheet数据
'=====================================================
Function ValidateSheetData(ws As Worksheet, wsData As Variant, dict As Object, wsOutput As Worksheet, _
                           ByRef outputRow As Long, lastRow As Long, subjectCol As Long, folderCol As Long, _
                           dateColIndices As Collection, dateColNames As Collection, colOffset As Long) As Long
    Dim i As Long, j As Long
    Dim subjectID As String
    Dim folderName As String
    Dim screeningD28Visit As Variant
    Dim screeningD7Visit As Variant
    Dim c1d1Visit As Variant
    Dim dateValue As Variant
    Dim actualDate As Date
    Dim hasErrors As Boolean
    Dim errorMsg As String
    Dim errorType As String
    Dim outputData() As Variant
    Dim outputCount As Long
    Dim maxOutputRows As Long
    Dim sheetErrorCount As Long

    sheetErrorCount = 0

    ' 准备输出数组
    maxOutputRows = lastRow * dateColIndices.Count
    ReDim outputData(1 To maxOutputRows, 1 To 11)
    outputCount = 0

    ' 验证每一行
    For i = 2 To lastRow
        subjectID = CleanString(Trim(CStr(wsData(i, subjectCol + colOffset))))
        If Len(subjectID) = 0 Then GoTo NextRow

        folderName = CleanString(Trim(CStr(wsData(i, folderCol + colOffset))))
        If Len(folderName) = 0 Then GoTo NextRow

        folderName = NormalizeVisitName(folderName)

        screeningD28Visit = dict(subjectID & "|Screening(-D28)")
        screeningD7Visit = dict(subjectID & "|Screening(-D7)")
        c1d1Visit = dict(subjectID & "|C1D1")

        ' 检查每个日期列
        For j = 1 To dateColIndices.Count
            Dim colIndex As Long
            colIndex = dateColIndices(j)

            dateValue = wsData(i, colIndex + colOffset)

            If IsDate(dateValue) Then
                actualDate = CDate(dateValue)
            ElseIf IsNumeric(dateValue) And dateValue > 0 Then
                actualDate = CDate(dateValue)
            Else
                GoTo NextDateColumn
            End If

            hasErrors = False

            ' 验证逻辑
            Select Case folderName
                Case "Screening(-D28)"
                    If IsDate(screeningD7Visit) Then
                        If actualDate > CDate(screeningD7Visit) Then
                            hasErrors = True
                            errorType = "Screening(-D28)数据日期晚于Screening(-D7)访视日期"
                            errorMsg = "访视Screening(-D28)的数据日期(" & Format(actualDate, "yyyy-mm-dd") & _
                                       ") 晚于Screening(-D7)的访视日期(" & Format(CDate(screeningD7Visit), "yyyy-mm-dd") & ")"
                        End If
                    End If

                    If IsDate(c1d1Visit) And Not hasErrors Then
                        If actualDate > CDate(c1d1Visit) Then
                            hasErrors = True
                            errorType = "Screening(-D28)数据日期晚于C1D1访视日期"
                            errorMsg = "访视Screening(-D28)的数据日期(" & Format(actualDate, "yyyy-mm-dd") & _
                                       ") 晚于C1D1的访视日期(" & Format(CDate(c1d1Visit), "yyyy-mm-dd") & ")"
                        End If
                    End If

                Case "Screening(-D7)"
                    If IsDate(c1d1Visit) Then
                        If actualDate > CDate(c1d1Visit) Then
                            hasErrors = True
                            errorType = "Screening(-D7)数据日期晚于C1D1访视日期"
                            errorMsg = "访视Screening(-D7)的数据日期(" & Format(actualDate, "yyyy-mm-dd") & _
                                       ") 晚于C1D1的访视日期(" & Format(CDate(c1d1Visit), "yyyy-mm-dd") & ")"
                        End If
                    End If
            End Select

            ' 记录错误
            If hasErrors Then
                outputCount = outputCount + 1

                outputData(outputCount, 1) = ws.Name
                outputData(outputCount, 2) = i
                outputData(outputCount, 3) = subjectID
                outputData(outputCount, 4) = folderName
                outputData(outputCount, 5) = dateColNames(j)
                outputData(outputCount, 6) = actualDate

                If IsDate(screeningD28Visit) Then
                    outputData(outputCount, 7) = CDate(screeningD28Visit)
                End If

                If IsDate(screeningD7Visit) Then
                    outputData(outputCount, 8) = CDate(screeningD7Visit)
                End If

                If IsDate(c1d1Visit) Then
                    outputData(outputCount, 9) = CDate(c1d1Visit)
                End If

                outputData(outputCount, 10) = errorType
                outputData(outputCount, 11) = errorMsg

                sheetErrorCount = sheetErrorCount + 1
            End If

NextDateColumn:
        Next j

NextRow:
    Next i

    ' 批量写入输出数据
    If outputCount > 0 Then
        wsOutput.Range(wsOutput.Cells(outputRow, 1), wsOutput.Cells(outputRow + outputCount - 1, 11)).Value = outputData

        Dim formatRange As Range
        Set formatRange = wsOutput.Range(wsOutput.Cells(outputRow, 6), wsOutput.Cells(outputRow + outputCount - 1, 9))
        formatRange.NumberFormat = "yyyy-mm-dd"

        outputRow = outputRow + outputCount
    End If

    ValidateSheetData = sheetErrorCount
End Function

Function FindColumnBilingual(data As Variant, colType As String, startCol As Long, endCol As Long) As Long
    Dim j As Long
    Dim cellValue As String
    Dim cleanValue As String
    Dim mapping As Object

    FindColumnBilingual = 0

    Select Case LCase(colType)
        Case "subject"
            Set mapping = subjectColMapping
        Case "folder"
            Set mapping = folderColMapping
        Case "visitdate"
            Set mapping = visitDateColMapping
        Case Else
            Exit Function
    End Select

    For j = startCol To endCol
        On Error Resume Next
        cellValue = CStr(data(1, j))
        If Err.Number = 0 Then
            cleanValue = LCase(CleanString(Trim(cellValue)))
            If mapping.Exists(cleanValue) Then
                FindColumnBilingual = j
                Exit Function
            End If
        End If
        On Error GoTo 0
    Next j
End Function

Function NormalizeVisitName(visitName As String) As String
    Dim cleanName As String
    cleanName = LCase(CleanString(Trim(visitName)))

    If visitNameMapping.Exists(cleanName) Then
        NormalizeVisitName = visitNameMapping(cleanName)
    Else
        NormalizeVisitName = CleanString(Trim(visitName))
    End If
End Function

'=====================================================
' 清理字符串函数
'=====================================================
Function CleanString(ByVal text As String) As String
    If Len(text) = 0 Then
        CleanString = ""
        Exit Function
    End If

    text = Replace(text, Chr(160), " ")
    text = Replace(text, ChrW(160), " ")
    text = Replace(text, Chr(9), " ")
    text = Replace(text, Chr(10), " ")
    text = Replace(text, Chr(13), " ")

    Dim pos As Long
    pos = InStr(text, "  ")
    Do While pos > 0
        text = Left(text, pos) & Mid(text, pos + 2)
        pos = InStr(text, "  ")
    Loop

    CleanString = Trim(text)
End Function

Function AnalyzeAndValidateSheet(ws As Worksheet, dict As Object, wsOutput As Worksheet, _
                                  ByRef outputRow As Long, ByRef errorCount As Long) As Variant
    Dim result(6) As Variant
    Dim lastRow As Long
    Dim lastCol As Long
    Dim wsData As Variant
    Dim subjectCol As Long
    Dim folderCol As Long
    Dim dateColIndices As Collection
    Dim dateColNames As Collection
    Dim j As Long
    Dim hasVisitColumns As Boolean
    Dim colOffset As Long
    Dim dateColList As String

    ' 获取数据范围
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    result(1) = lastRow - 1
    result(2) = lastCol

    If lastRow < 2 Then
        result(0) = "未检查：无数据行"
        result(3) = "-"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 批量读取数据
    On Error Resume Next
    wsData = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, lastCol)).Value
    If Err.Number <> 0 Then
        result(0) = "未检查：读取错误"
        result(3) = Err.Description
        AnalyzeAndValidateSheet = result
        Exit Function
    End If
    On Error GoTo 0

    ' 查找Subject列
    subjectCol = FindColumnBilingual(wsData, "subject", 1, lastCol)
    If subjectCol = 0 Then
        result(0) = "未检查：未找到Subject列"
        result(3) = "未找到：Subject/受试者代码/受试者姓名或标识等列名"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 查找Folder列
    folderCol = FindColumnBilingual(wsData, "folder", 1, lastCol)
    If folderCol = 0 Then
        result(0) = "未检查：未找到Folder列"
        result(3) = "未找到：Folder instance name/访视名称等列名"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 检查是否已添加访视日期列（检查前三列是否包含Screening和Date）
    hasVisitColumns = False
    If lastCol >= 3 Then
        Dim col1Name As String, col2Name As String, col3Name As String
        col1Name = CleanString(Trim(CStr(wsData(1, 1))))
        col2Name = CleanString(Trim(CStr(wsData(1, 2))))
        col3Name = CleanString(Trim(CStr(wsData(1, 3))))
        
        If InStr(col1Name, "Screening") > 0 And InStr(col1Name, "Date") > 0 Then
            hasVisitColumns = True
        End If
    End If
    
    If hasVisitColumns Then
        colOffset = 3
    Else
        colOffset = 0
    End If

    ' 查找日期列
    Set dateColIndices = New Collection
    Set dateColNames = New Collection

    For j = 1 To lastCol
        If hasVisitColumns And j <= 3 Then GoTo NextColumn

        Dim colName As String
        colName = CleanString(Trim(CStr(wsData(1, j))))

        Dim hasDateKeyword As Boolean
        hasDateKeyword = False
        
        Dim key As Variant
        For Each key In dateColKeywords.Keys
            If InStr(1, colName, key, vbTextCompare) > 0 Then
                If InStr(1, LCase(colName), "update") > 0 Or _
                   InStr(1, LCase(colName), "creation") > 0 Or _
                   InStr(1, LCase(colName), "更新") > 0 Or _
                   InStr(1, LCase(colName), "创建") > 0 Then
                    hasDateKeyword = False
                Else
                    hasDateKeyword = True
                    Exit For
                End If
            End If
        Next

        If hasDateKeyword Then
            dateColIndices.Add j
            dateColNames.Add colName
        End If

NextColumn:
    Next j

    If dateColIndices.Count = 0 Then
        result(0) = "未检查：未找到日期列"
        result(3) = "未找到包含Date/日期关键词的列（已排除update/creation等列）"
        AnalyzeAndValidateSheet = result
        Exit Function
    End If

    ' 构建日期列列表
    dateColList = ""
    For j = 1 To WorksheetFunction.Min(dateColNames.Count, 3)
        If j > 1 Then dateColList = dateColList & ", "
        dateColList = dateColList & dateColNames(j)
    Next j
    If dateColNames.Count > 3 Then
        dateColList = dateColList & " ... (共" & dateColNames.Count & "列)"
    End If

    ' 执行验证
    Dim sheetErrorCount As Long
    sheetErrorCount = ValidateSheetData(ws, wsData, dict, wsOutput, outputRow, _
                                        lastRow, subjectCol, folderCol, dateColIndices, dateColNames, colOffset)

    ' 返回结果
    result(0) = "已检查"
    result(2) = "列" & subjectCol & ": " & CleanString(Trim(CStr(wsData(1, subjectCol))))
    result(3) = "列" & folderCol & ": " & CleanString(Trim(CStr(wsData(1, folderCol))))
    result(4) = dateColList
    result(5) = "已完成检查"
    result(6) = sheetErrorCount

    AnalyzeAndValidateSheet = result
End Function

Function ValidateSheetData(ws As Worksheet, wsData As Variant, dict As Object, wsOutput As Worksheet, _
                           ByRef outputRow As Long, lastRow As Long, subjectCol As Long, folderCol As Long, _
                           dateColIndices As Collection, dateColNames As Collection, colOffset As Long) As Long
    Dim i As Long, j As Long
    Dim subjectID As String
    Dim folderName As String
    Dim screeningD28Visit As Variant
    Dim screeningD7Visit As Variant
    Dim c1d1Visit As Variant
    Dim dateValue As Variant
    Dim actualDate As Date
    Dim hasErrors As Boolean
    Dim errorMsg As String
    Dim errorType As String
    Dim outputData() As Variant
    Dim outputCount As Long
    Dim maxOutputRows As Long
    Dim sheetErrorCount As Long

    sheetErrorCount = 0

    ' 准备输出数组
    maxOutputRows = lastRow * dateColIndices.Count
    ReDim outputData(1 To maxOutputRows, 1 To 11)
    outputCount = 0

    ' 验证每一行
    For i = 2 To lastRow
        subjectID = CleanString(Trim(CStr(wsData(i, subjectCol + colOffset))))
        If Len(subjectID) = 0 Then GoTo NextRow

        folderName = CleanString(Trim(CStr(wsData(i, folderCol + colOffset))))
        If Len(folderName) = 0 Then GoTo NextRow

        folderName = NormalizeVisitName(folderName)

        screeningD28Visit = dict(subjectID & "|Screening(-D28)")
        screeningD7Visit = dict(subjectID & "|Screening(-D7)")
        c1d1Visit = dict(subjectID & "|C1D1")

        ' 检查每个日期列
        For j = 1 To dateColIndices.Count
            Dim colIndex As Long
            colIndex = dateColIndices(j)

            dateValue = wsData(i, colIndex + colOffset)

            If IsDate(dateValue) Then
                actualDate = CDate(dateValue)
            ElseIf IsNumeric(dateValue) And dateValue > 0 Then
                actualDate = CDate(dateValue)
            Else
                GoTo NextDateColumn
            End If

            hasErrors = False

            ' 验证逻辑
            Select Case folderName
                Case "Screening(-D28)"
                    If IsDate(screeningD7Visit) Then
                        If actualDate > CDate(screeningD7Visit) Then
                            hasErrors = True
                            errorType = "Screening(-D28)数据日期晚于Screening(-D7)访视日期"
                            errorMsg = "访视Screening(-D28)的数据日期(" & Format(actualDate, "yyyy-mm-dd") & _
                                       ") 晚于Screening(-D7)的访视日期(" & Format(CDate(screeningD7Visit), "yyyy-mm-dd") & ")"
                        End If
                    End If

                    If IsDate(c1d1Visit) And Not hasErrors Then
                        If actualDate > CDate(c1d1Visit) Then
                            hasErrors = True
                            errorType = "Screening(-D28)数据日期晚于C1D1访视日期"
                            errorMsg = "访视Screening(-D28)的数据日期(" & Format(actualDate, "yyyy-mm-dd") & _
                                       ") 晚于C1D1的访视日期(" & Format(CDate(c1d1Visit), "yyyy-mm-dd") & ")"
                        End If
                    End If

                Case "Screening(-D7)"
                    If IsDate(c1d1Visit) Then
                        If actualDate > CDate(c1d1Visit) Then
                            hasErrors = True
                            errorType = "Screening(-D7)数据日期晚于C1D1访视日期"
                            errorMsg = "访视Screening(-D7)的数据日期(" & Format(actualDate, "yyyy-mm-dd") & _
                                       ") 晚于C1D1的访视日期(" & Format(CDate(c1d1Visit), "yyyy-mm-dd") & ")"
                        End If
                    End If
            End Select

            ' 记录错误
            If hasErrors Then
                outputCount = outputCount + 1

                outputData(outputCount, 1) = ws.Name
                outputData(outputCount, 2) = i
                outputData(outputCount, 3) = subjectID
                outputData(outputCount, 4) = folderName
                outputData(outputCount, 5) = dateColNames(j)
                outputData(outputCount, 6) = actualDate

                If IsDate(screeningD28Visit) Then
                    outputData(outputCount, 7) = CDate(screeningD28Visit)
                End If

                If IsDate(screeningD7Visit) Then
                    outputData(outputCount, 8) = CDate(screeningD7Visit)
                End If

                If IsDate(c1d1Visit) Then
                    outputData(outputCount, 9) = CDate(c1d1Visit)
                End If

                outputData(outputCount, 10) = errorType
                outputData(outputCount, 11) = errorMsg

                sheetErrorCount = sheetErrorCount + 1
            End If

NextDateColumn:
        Next j

NextRow:
    Next i

    ' 批量写入输出数据
    If outputCount > 0 Then
        wsOutput.Range(wsOutput.Cells(outputRow, 1), wsOutput.Cells(outputRow + outputCount - 1, 11)).Value = outputData

        Dim formatRange As Range
        Set formatRange = wsOutput.Range(wsOutput.Cells(outputRow, 6), wsOutput.Cells(outputRow + outputCount - 1, 9))
        formatRange.NumberFormat = "yyyy-mm-dd"

        outputRow = outputRow + outputCount
    End If

    ValidateSheetData = sheetErrorCount
End Function

Function FindColumnBilingual(data As Variant, colType As String, startCol As Long, endCol As Long) As Long

Function CleanString(ByVal text As String) As String
    If Len(text) = 0 Then
        CleanString = ""
        Exit Function
    End If

    text = Replace(text, Chr(160), " ")
    text = Replace(text, ChrW(160), " ")
    text = Replace(text, Chr(9), " ")
    text = Replace(text, Chr(10), " ")
    text = Replace(text, Chr(13), " ")

    Dim pos As Long
    pos = InStr(text, "  ")
    Do While pos > 0
        text = Left(text, pos) & Mid(text, pos + 2)
        pos = InStr(text, "  ")
    Loop

    CleanString = Trim(text)
End Function

Function GetVisitConfigByVisitName(visitName As String) As VisitConfig
    Dim i As Long
    Dim cleanVisitName As String

    cleanVisitName = CleanString(Trim(LCase(visitName)))

    For i = 1 To VisitCount
        ' 检查英文名称
        If LCase(CleanString(Visits(i).VisitNameEN)) = cleanVisitName Then
            GetVisitConfigByVisitName = Visits(i)
            Exit Function
        End If

        ' 检查中文名称
        If LCase(CleanString(Visits(i).VisitNameCN)) = cleanVisitName Then
            GetVisitConfigByVisitName = Visits(i)
            Exit Function
        End If

        ' 检查别名
        Dim aliases() As String
        aliases = Split(Visits(i).VisitNameAliases, ",")
        Dim j As Long
        For j = LBound(aliases) To UBound(aliases)
            If LCase(CleanString(Trim(aliases(j)))) = cleanVisitName Then
                GetVisitConfigByVisitName = Visits(i)
                Exit Function
            End If
        Next j
    Next i

    ' 未找到，返回空配置
    GetVisitConfigByVisitName.visitName = ""
End Function

'=====================================================
' 辅助函数：清理字符串
'=====================================================
Function CleanString(ByVal text As String) As String
    If Len(text) = 0 Then
        CleanString = ""
        Exit Function
    End If

    text = Replace(text, Chr(160), " ")
    text = Replace(text, ChrW(160), " ")
    text = Replace(text, Chr(9), " ")
    text = Replace(text, Chr(10), " ")
    text = Replace(text, Chr(13), " ")

    Dim pos As Long
    pos = InStr(text, "  ")
    Do While pos > 0
        text = Left(text, pos) & Mid(text, pos + 2)
        pos = InStr(text, "  ")
    Loop

    CleanString = Trim(text)
End Function

'=====================================================
' 辅助函数：根据访视名称查找访视配置
'=====================================================
Function GetVisitConfigByVisitName(visitName As String) As VisitConfig
    Dim i As Long
    Dim cleanVisitName As String
    Dim defaultConfig As VisitConfig

    cleanVisitName = CleanString(Trim(LCase(visitName)))
    
    ' 确保Visits数组已初始化
    If VisitCount = 0 Then
        Call InitializeVisitConfiguration
    End If

    For i = 1 To VisitCount
        ' 只检查启用的访视
        If Visits(i).IsEnabled Then
            ' 检查英文名称
            If LCase(CleanString(Visits(i).VisitNameEN)) = cleanVisitName Then
                GetVisitConfigByVisitName = Visits(i)
                Exit Function
            End If

            ' 检查中文名称
            If Len(Visits(i).VisitNameCN) > 0 Then
                If LCase(CleanString(Visits(i).VisitNameCN)) = cleanVisitName Then
                    GetVisitConfigByVisitName = Visits(i)
                    Exit Function
                End If
            End If

            ' 检查别名
            If Len(Visits(i).VisitNameAliases) > 0 Then
                Dim aliases() As String
                aliases = Split(Visits(i).VisitNameAliases, ",")
                Dim j As Long
                For j = LBound(aliases) To UBound(aliases)
                    If LCase(CleanString(Trim(aliases(j)))) = cleanVisitName Then
                        GetVisitConfigByVisitName = Visits(i)
                        Exit Function
                    End If
                Next j
            End If
        End If
    Next i

    ' 未找到，返回空配置
    defaultConfig.visitName = ""
    defaultConfig.IsEnabled = False
    GetVisitConfigByVisitName = defaultConfig
End Function

'=====================================================
' 辅助函数：清理字符串
'=====================================================
Function CleanString(ByVal text As String) As String
    If Len(text) = 0 Then
        CleanString = ""
        Exit Function
    End If

    text = Replace(text, Chr(160), " ")
    text = Replace(text, ChrW(160), " ")
    text = Replace(text, Chr(9), " ")
    text = Replace(text, Chr(10), " ")
    text = Replace(text, Chr(13), " ")

    Dim pos As Long
    pos = InStr(text, "  ")
    Do While pos > 0
        text = Left(text, pos) & Mid(text, pos + 2)
        pos = InStr(text, "  ")
    Loop

    CleanString = Trim(text)
End Function
